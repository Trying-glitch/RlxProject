local cachedToken = nil
local lastTokenTick = 0
local function generateToken(currentKey)
	if not currentKey or not Stellify.ZX_Parry.TransformFn then
		return nil
	end
	if tick() - lastTokenTick < 0.015 and cachedToken then
		return cachedToken
	end
	local tok, transformed = pcall(Stellify.ZX_Parry.TransformFn, currentKey, "TIME")
	if not tok or not transformed then
		return nil
	end
	local serverTime = workspace:GetServerTimeNow() * 100
	local timeStr = tostring(math.floor(serverTime))
	local tokenChars = {}
	for i = 1, # timeStr do
		local ki = (i - 1) % # transformed + 1
		local xb = bit32.bxor((string.byte(timeStr, i) + i) % 256, string.byte(transformed, ki))
		tokenChars[i] = string.char(xb)
	end
	cachedToken = table.concat(tokenChars)
	lastTokenTick = tick()
	return cachedToken
end

		return Closest_Entity
	end

	if not Alive then

	end
	for _, entity in pairs(Alive:GetChildren()) do

			local distance = LocalPlayer:DistanceFromCharacter(entity.PrimaryPart.Position)

function Stellify.player.get_closest_to_cursor()
	if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild('HumanoidRootPart') then
		return nil
	end
	local closest_player, minimal_dot = nil, - math.huge
	local camera = workspace.CurrentCamera
	if not Alive then
		return nil
	end
	local success, mouse_location = pcall(function()
		return UserInputService:GetMouseLocation()
	end)
	if not success then
		return nil
	end
	local ray = camera:ScreenPointToRay(mouse_location.X, mouse_location.Y)
	local pointer = CFrame.lookAt(ray.Origin, ray.Origin + ray.Direction)
	for _, player in pairs(Alive:GetChildren()) do
		if player == LocalPlayer.Character or not player:FindFirstChild('HumanoidRootPart') then
			continue
		end
		local direction = (player.HumanoidRootPart.Position - camera.CFrame.Position).Unit
		local dot = pointer.LookVector:Dot(direction)
		if dot > minimal_dot then
			minimal_dot = dot
			closest_player = player
		end
	end
	return closest_player
end

function Stellify.curve.get_cframe()
	local camera = workspace.CurrentCamera
	local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild('HumanoidRootPart')
	if not root then
		return camera.CFrame
	end
	local targetPart = nil
	local closest = Stellify.player.get_closest_to_cursor()
	if closest and closest:FindFirstChild('HumanoidRootPart') then
		targetPart = closest.HumanoidRootPart
	end
	local target_pos = targetPart and targetPart.Position or (root.Position + camera.CFrame.LookVector * 100)
	local curve_functions = {
		[1] = function()
			return camera.CFrame
		end,
		[2] = function()
			local direction, random_offset, attempts = (target_pos - root.Position).Unit, Vector3.new(), 0
			repeat
				random_offset = Vector3.new(math.random(- 4000, 4000), math.random(- 4000, 4000), math.random(- 4000, 4000))
				attempts = attempts + 1
			until direction:Dot((target_pos + random_offset - root.Position).Unit) < 0.95 or attempts > 10
			return CFrame.new(root.Position, target_pos + random_offset)
		end,
		[3] = function()
			return CFrame.new(root.Position, target_pos + Vector3.new(0, 5, 0))
		end,
		[4] = function()
			return CFrame.new(camera.CFrame.Position, root.Position + (root.Position - target_pos).Unit * 10000 + Vector3.new(0, 1000, 0))
		end,
		[5] = function()
			return CFrame.new(root.Position, target_pos + Vector3.new(0, - 9e18, 0))
		end,
		[6] = function()
			return CFrame.new(root.Position, target_pos + Vector3.new(0, 9e18, 0))
		end,
		[7] = function()
			local candidates = {}
			if Alive then
				for _, pl in pairs(Alive:GetChildren()) do
					if pl ~= LocalPlayer.Character and pl.PrimaryPart then
						table.insert(candidates, pl)
					end
				end
			end
			if # candidates > 0 then
				return CFrame.new(root.Position, candidates[math.random(1, # candidates)].PrimaryPart.Position)
			end
			return camera.CFrame
		end,
		[8] = function()
			return CFrame.new(root.Position, root.Position + (- camera.CFrame.RightVector * 10000))
		end,
		[9] = function()
			return CFrame.new(root.Position, root.Position + (camera.CFrame.RightVector * 10000))
		end
	}
	local selected_func = curve_functions[Stellify.__properties.__curve_mode] or curve_functions[1]
	return selected_func()
end

--trajectory and detection
Stellify.detection = {
	__ball_properties = {
		__aerodynamic_time = tick(),
		__last_warping = tick(),
		__lerp_radians = 0,
		__curving = tick()
	}
}
function Stellify.detection.is_curved()
	local props = Stellify.detection.__ball_properties
	local ball = Stellify.ball.get()
	if not ball then
		return false
	end
	local zoomies = ball:FindFirstChild("zoomies")
	local velocity = zoomies and zoomies.VectorVelocity or ball.AssemblyLinearVelocity
	local speed = velocity.Magnitude
	if speed < 5 then
		return false
	end
	if # _ZX_VelHistory.ball >= 3 then
		local p1 = _ZX_VelHistory.ball[1].pos
		local p2 = _ZX_VelHistory.ball[2].pos
		local p3 = _ZX_VelHistory.ball[3].pos
		local deviation = math.acos(math.clamp((p1 - p2).Unit:Dot((p2 - p3).Unit), - 1, 1))
		if deviation > 0.032 or ball:FindFirstChild("AeroDynamicSlashVFX") then
			return true
		end
	end
	local char = LocalPlayer.Character
	if not char or not char.PrimaryPart then
		return false
	end
	local pos = char.PrimaryPart.Position
	local direction = (pos - ball.Position).Unit
	local dot = direction:Dot(velocity.Unit)
	if dot < 0.22 then
		return true
	end
	local ping = Stats.Network.ServerStatsItem:GetValue() / 1000
	local distance = (pos - ball.Position).Magnitude
	local reach_time = distance / speed - ping
	local dot_threshold = math.clamp(0.55 - (ping * 0.75), - 1, 0.45)
	local ball_distance_threshold = 15 - math.min(distance / 1000, 15) + math.min(speed / 100, 45)
	local radians = math.asin(math.clamp(dot, - 1, 1))
	props.__lerp_radians = props.__lerp_radians + (radians - props.__lerp_radians) * 0.85
	if props.__lerp_radians < 0.016 then
		props.__last_warping = tick()
	end
	if distance < (ball_distance_threshold * 0.85) then
		return false
	end
	if (tick() - props.__last_warping) < (reach_time / 1.4) or (tick() - props.__curving) < (reach_time / 1.1) then
		return true
	end
	return dot < dot_threshold
end

--triggerbot
Stellify.triggerbot = {}
function Stellify.triggerbot.trigger(ball)
	if Stellify.__triggerbot.__is_parrying or Stellify.__triggerbot.__parries > Stellify.__triggerbot.__max_parries then
		return
	end
	if LocalPlayer.Character and LocalPlayer.Character.PrimaryPart and LocalPlayer.Character.PrimaryPart:FindFirstChild('SingularityCape') then
		return
	end
	Stellify.__triggerbot.__is_parrying = true
	Stellify.__triggerbot.__parries = Stellify.__triggerbot.__parries + 1
	Stellify.animation.play_grab_parry()
	Stellify.parry.execute()
	task.delay(Stellify.__triggerbot.__parry_delay, function()
		Stellify.__triggerbot.__parries = math.max(0, Stellify.__triggerbot.__parries - 1)
	end)
	local connection
	connection = ball:GetAttributeChangedSignal('target'):Once(function()
		Stellify.__triggerbot.__is_parrying = false;
		if connection then
			connection:Disconnect()
		end
	end)
	task.spawn(function()
		local start_time = tick()
		repeat
			RunService.Heartbeat:Wait()
		until (tick() - start_time >= 1 or not Stellify.__triggerbot.__is_parrying)
		Stellify.__triggerbot.__is_parrying = false
	end)
end
function Stellify.triggerbot.loop()
	if not Stellify.__triggerbot.__enabled then
		return
	end
	local balls = workspace:FindFirstChild('Balls')
	if not balls then
		return
	end
	for _, ball in pairs(balls:GetChildren()) do
		if ball:IsA('BasePart') and ball:GetAttribute('target') == LocalPlayer.Name then
			Stellify.triggerbot.trigger(ball)
			break
		end
	end
end
function Stellify.triggerbot.enable(enabled)
	Stellify.__triggerbot.__enabled = enabled
	if enabled then
		if not Stellify.__properties.__connections.__triggerbot then
			Stellify.__properties.__connections.__triggerbot = RunService.Heartbeat:Connect(Stellify.triggerbot.loop)
		end
	else
		if Stellify.__properties.__connections.__triggerbot then
			Stellify.__properties.__connections.__triggerbot:Disconnect();
			Stellify.__properties.__connections.__triggerbot = nil
		end
		Stellify.__triggerbot.__is_parrying = false
		Stellify.__triggerbot.__parries = 0
	end
end

-- unified engine (Spam execution zero CPU overhead rebuild)
Stellify.autoparry = {}
local parryFlag = false
local isExecutingSlashes = false
local isExecutingSpam = false
function Stellify.autoparry.start()
	if Stellify.__properties.__connections.__combat then
		Stellify.__properties.__connections.__combat:Disconnect()
	end
	parryFlag = false
	isExecutingSlashes = false
	isExecutingSpam = false
	Stellify.__properties.__connections.__combat = RunService.PreSimulation:Connect(function(dt)
		if not Stellify.__properties.__autoparry_enabled or not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then
			return
		end
		
		-- auto abilities
		if getgenv().AutoAbility or (Stellify and Stellify.__properties and Stellify.__properties.__auto_ability_enabled) then
			local hotbar = LocalPlayer.PlayerGui:FindFirstChild("Hotbar")
			local abilityGui = hotbar and hotbar:FindFirstChild("Ability")
			local AbilityCD = abilityGui and abilityGui:FindFirstChild("UIGradient")
			local char = LocalPlayer.Character
			local abs = char and char:FindFirstChild("Abilities")
			if abs then
				local ball = Stellify.ball.get()
				if ball and ball.Parent then
					local currentTarget = ball:GetAttribute("target")
					local isTargeted = (currentTarget == LocalPlayer.Name)
					local distance = LocalPlayer:DistanceFromCharacter(ball.Position)
					local isAbilityReady = false
					pcall(function()
						isAbilityReady = AbilityCD and math.abs(AbilityCD.Offset.Y - 0.5) < 0.05
					end)
					local currentParryCount = Stellify.__properties.__parries or 0
					local slashes = abs:FindFirstChild("Slashes Of Fury") or abs:FindFirstChild("Slashes of Fury")
					local isSlashesReady = slashes and slashes.Enabled and isAbilityReady
					if isSlashesReady and isTargeted then
						parryFlag = true
						if distance <= 35 and not isExecutingSlashes then
							task.spawn(function()
								isExecutingSlashes = true
								pcall(function()
									local btn = ReplicatedStorage.Remotes:FindFirstChild("AbilityButtonPress")
									if btn then
										if btn:IsA("RemoteEvent") then
											btn:FireServer()
										else
											btn:Fire()
										end
									end
								end)
								
								-- [FIXED]: Perfectly count exactly 34 parries and pace them smoothly
								local maxSlashes = 34
								local currentSlashes = 0
								while currentSlashes < maxSlashes do
									Stellify.parry.execute_bruteforce()
									currentSlashes = currentSlashes + 1
									task.wait(0.025)
								end
								task.wait(0.5) -- Buffer before resetting execution state
								parryFlag = false
								isExecutingSlashes = false
							end)
						end
					elseif isAbilityReady and isTargeted and (currentParryCount > 1) then
						local standardAbilities = {
							"Raging Deflection",
							"Rapture",
							"Calming Deflection",
							"Aerodynamic Slash",
							"Fracture",
							"Phantom"
						}
						for _, abilityName in ipairs(standardAbilities) do
							local abilityObj = abs:FindFirstChild(abilityName)
							if abilityObj and abilityObj.Enabled and distance <= 45 then
								if Stellify and Stellify.__properties then
									Stellify.__properties.__parried = true
								end
								parryFlag = true
								pcall(function()
									local btn = ReplicatedStorage.Remotes:FindFirstChild("AbilityButtonPress")
									if btn then
										if btn:IsA("RemoteEvent") then
											btn:FireServer()
										else
											btn:Fire()
										end
									end
								end)
								task.delay(0.75, function()
									parryFlag = false
								end)
								break
							end
						end
					end
				end
			end
		end

		-- parry and spam logic
		local balls = Stellify.ball.get_all()
		for _, ball in pairs(balls) do
			local zoomies = ball:FindFirstChild('zoomies')
			if not zoomies then
				continue
			end
			if not ball:GetAttribute("Stellify_TargetTracked") then
				ball:SetAttribute("Stellify_TargetTracked", true)
				local targetConn
				targetConn = ball:GetAttributeChangedSignal("target"):Connect(function()
					parryFlag = false
				end)
				local destroyConn
				destroyConn = ball.Destroying:Connect(function()
					if targetConn then
						targetConn:Disconnect()
					end
					if destroyConn then
						destroyConn:Disconnect()
					end
				end)
			end
			local localName = LocalPlayer.Name
			local currentTarget = ball:GetAttribute('target')
			local isTargeted = currentTarget == localName
			local velocity = zoomies.VectorVelocity
			local ballPos = ball.Position
			local playerPos = LocalPlayer.Character.PrimaryPart.Position
			local distance = (playerPos - ballPos).Magnitude
			table.insert(_ZX_VelHistory.ball, 1, {
				pos = ballPos,
				vel = velocity,
				t = tick()
			})
			if # _ZX_VelHistory.ball > _ZX_VelHistory.MAX_SAMPLES then
				table.remove(_ZX_VelHistory.ball, # _ZX_VelHistory.ball)
			end
			local ballSpeed = math.max(velocity.Magnitude, 0)
			local pingValue = 0
			pcall(function()
				pingValue = Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
			end)
			local ping = pingValue / 1000
			local spamThresh = Stellify.__properties.__spam_threshold * Stellify.__properties.__auto_spam_distance_multiplier
			local currentParryCount = Stellify.__properties.__parries or 0
			local manualHold = Stellify.__properties.__manual_spam_enabled and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
			local autoSpamConditionsMet = Stellify.__properties.__auto_spam_enabled and (currentParryCount > 1)
			
			-- Static wait handler defined outside execution loop for maximum performance
			local function executeSpamWait()
				if Stellify.__properties.__spam_wait_method == "Maximum Game Stability" then
					task.wait()
				else
					RunService.PreSimulation:Wait()
				end
			end

						-- ZERO OVERHEAD SPAM CORE FIX
			if (autoSpamConditionsMet or manualHold) and distance <= spamThresh then
				if not isExecutingSpam then
					isExecutingSpam = true
					Stellify.animation.play_grab_parry()
				
				-- Calculate optimal batch amount for this frame
					local maxBatch = (Stellify.__properties.__spam_batch_amount == "Fast Speed") and 33 or 45
					local currentBatchAmount = maxBatch
					if Stellify.__properties.__randomized_spam_enabled then
						local pingFactor = 1 + math.clamp(ping, 0, 0.5)
						local speedFactor = math.floor(ballSpeed / 40)
						local optimalBatch = math.clamp(math.floor((speedFactor + (Stellify.__properties.__spam_rate / 12)) * pingFactor), 2, maxBatch)
						currentBatchAmount = math.random(math.max(1, optimalBatch - 2), optimalBatch)
					else
						local baseSpam = math.floor(ballSpeed / 50)
						currentBatchAmount = math.clamp(math.floor((baseSpam + (Stellify.__properties.__spam_rate / 15))), 1, maxBatch)
					end
				
				-- Native batch execution utilizing executeSpamWait()
					if Stellify.ZX_Parry.Hooked and Stellify.ZX_Parry.Remote then
						local keyIndex = Stellify.ZX_Parry.KeyTable and Stellify.ZX_Parry.KeyTable[3]
						local currentKey = keyIndex and Stellify.ZX_Parry.KeyTable[1][keyIndex]
						if currentKey then
							local token = generateToken(currentKey)
							local pCF = Stellify.curve.get_cframe()
							if token then
								local cam = workspace.CurrentCamera
							-- Cache screen positions once per frame window
								if tick() - Cache_Update_Tick > 0.1 then
									table.clear(Last_Positions_Cache)
									if Alive and cam then
										for _, entity in ipairs(Alive:GetChildren()) do
											if entity.PrimaryPart and entity.Name ~= localName then
												local ok, sp = pcall(cam.WorldToScreenPoint, cam, entity.PrimaryPart.Position)
												if ok then
													Last_Positions_Cache[entity.Name] = sp
												end
											end
										end
									end
									Cache_Update_Tick = tick()
								end
								local vpX, vpY = cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2
								local vpCenter = {
									vpX,
									vpY
								}
								task.spawn(function()
									for _ = 1, currentBatchAmount do
										pcall(function()
											Stellify.ZX_Parry.Remote:FireServer(Stellify.ZX_Parry.ParryHash, currentKey, token, 0.5, pCF, Last_Positions_Cache, vpCenter, false)
										end)
										executeSpamWait()
									end
									isExecutingSpam = false -- [ADDED] Release lock once the batch finishes
								end)
								Stellify.__properties.__parries = Stellify.__properties.__parries + currentBatchAmount
								task.delay(0.5, function()
									Stellify.__properties.__parries = math.max(0, Stellify.__properties.__parries - currentBatchAmount)
								end)
							else
								isExecutingSpam = false -- [ADDED] Release lock if token generation failed
							end
						else
							isExecutingSpam = false -- [ADDED] Release lock if key extraction failed
						end
					else
					-- Fallback batch executor
						task.spawn(function()
							for _ = 1, currentBatchAmount do
								Stellify.parry.execute_bruteforce()
								executeSpamWait()
							end
							isExecutingSpam = false -- [ADDED] Release lock once bruteforce batch finishes
						end)
					end
				end
			end
			if not isTargeted or parryFlag then
				continue
			end
			local approachSpeed = velocity:Dot((playerPos - ballPos).Unit)
			local tti = (approachSpeed > 0) and (distance / approachSpeed) or math.huge
			local isCurved = Stellify.detection.is_curved()
			local reactionWindow = 0.35 + (ping * 1.5)
			local distanceThreshold = 15 + (Stellify.__properties.__accuracy / 10)
			if isCurved then
				local curveFactor = (ballSpeed > 0) and math.clamp(approachSpeed / ballSpeed, 0, 1) or 1
				local delayMultiplier = math.clamp(0.4 + (curveFactor * 0.35), 0.4, 0.85)
				reactionWindow = reactionWindow * delayMultiplier
				distanceThreshold = math.clamp(distanceThreshold * 0.4, 5, 10)
			else
				reactionWindow = reactionWindow + math.clamp(ballSpeed / 800, 0, 0.12)
			end
			if tti <= reactionWindow or distance <= distanceThreshold then
				if getgenv().AutoParryMode == "Keypress" then
					Stellify.parry.keypress()
				else
					Stellify.parry.execute_action()
				end
				parryFlag = true
				local lastCycle = tick()
				task.spawn(function()
					repeat
						RunService.PreSimulation:Wait()
					until (tick() - lastCycle) >= 1 or not parryFlag
					parryFlag = false
				end)
			end
		end
	end)
end

	if Stellify.__properties.__connections.__combat then
		Stellify.__properties.__connections.__combat:Disconnect()
		Stellify.__properties.__connections.__combat = nil
	end
end

			end
		end)
		if effectScripts then
