function isValidRemoteArgs(args)
	return # args == 7 and type(args[2]) == "string" and type(args[3]) == "number" and typeof(args[4]) == "CFrame" and type(args[5]) == "table" and type(args[6]) == "table" and type(args[7]) == "boolean"
end
function hookRemote(remote)
	if hookedRemotes[remote] then
		return
	end
	local mt = getrawmetatable(remote)
	if hookedMetatables[mt] then
		return
	end
	hookedMetatables[mt] = true
	setreadonly(mt, false)
	local originalIndex = mt.__index
	mt.__index = function(self, key)
		if (key == "FireServer" and self:IsA("RemoteEvent")) or (key == "InvokeServer" and self:IsA("RemoteFunction")) then
			return function(selfArg, ...)
				local args = {
					...
				}
				if isValidRemoteArgs(args) and not hookedRemotes[self] then
					hookedRemotes[self] = args
					lastActionName = args[2]
				end
				return originalIndex(self, key)(selfArg, unpack(args))
			end
		end
		return originalIndex(self, key)
	end
	setreadonly(mt, true)
end

function Util.getCurveCFrame()
	local camera = workspace.CurrentCamera
	local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not hrp then
		return camera.CFrame
	end
	local method = _G.config.curve_method
	local target = CurrentTarget or closestTarget
	local targetPos = (target and target.PrimaryPart and target.PrimaryPart.Position) or (hrp.Position + camera.CFrame.LookVector * 1000)
	local toTarget = (targetPos - hrp.Position).Unit
	if method == "dot" then
		return CFrame.lookAt(hrp.Position, targetPos + Vector3.new(0, 1.75, 0))
	elseif method == "backwards" then
		return CFrame.new(hrp.Position, hrp.Position + (- toTarget) * 1000)
	elseif method == "slow" then
		return CFrame.new(hrp.Position, hrp.Position + Vector3.new(0, - 350, 0))
	elseif method == "random" then
		return CFrame.new(hrp.Position, targetPos + Vector3.new(
            math.random(- 1000, 1000), math.random(- 350, 1000), math.random(- 1000, 1000)))
	else
		return camera.CFrame
	end
end

function Util.parry()
	local curveCFrame = Util.getCurveCFrame()
	local mouseLocation = UserInputService:GetMouseLocation()
	local mouseScreenPos = {
		mouseLocation.X,
		mouseLocation.Y
	}
	local playerScreenPositions = {}
	for _, character in workspace.Alive:GetChildren() do
		if character.PrimaryPart then
			local ok, screenPos = pcall(function()
				return workspace.CurrentCamera:WorldToScreenPoint(character.PrimaryPart.Position)
			end)
			if ok then
				playerScreenPositions[character.Name] = screenPos
			end
		end
	end
	local cursorOverride
	local cursorPos
	if cursorOverride then
		cursorPos = cursorOverride
	elseif isTouchDevice then
		local vpSize = workspace.CurrentCamera.ViewportSize
		cursorPos = {
			vpSize.X / 2,
			vpSize.Y / 2
		}
	else
		cursorPos = mouseScreenPos
	end
	for remote, originalArgs in hookedRemotes do
		local modifiedArgs = {
			originalArgs[1],
			originalArgs[2],
			originalArgs[3],
			curveCFrame,
			playerScreenPositions,
			cursorPos,
			originalArgs[7],
		}
		if remote:IsA("RemoteEvent") then
			remote:FireServer(unpack(modifiedArgs))
		elseif remote:IsA("RemoteFunction") then
			remote:InvokeServer(unpack(modifiedArgs))
		end
	end
	if parryCount > 7 then
		return false
	end
	parryCount += 1
	task.delay(0.5, function()
		if parryCount > 0 then
			parryCount -= 1
		end
	end)
end
function Util.ball_curved()
	local ballProps = Util.ball.properties
	local ball = Util.get_ball()
	if not ball then
		return false
	end
	local zoomies = ball:FindFirstChild("zoomies")
	if not zoomies then
		return false
	end
	local velocity = zoomies.VectorVelocity
	local velocityDir = velocity.Unit
	local toPlayer = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Unit
	local dotProduct = toPlayer:Dot(velocityDir)
	local speed = velocity.Magnitude
	local speedBonus = math.min(speed / 100, 40)
	local deviationDir = (velocityDir - velocity).Unit
	local deviationDot = toPlayer:Dot(deviationDir)
	local curveFactor = dotProduct - deviationDot
	local distToBall = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Magnitude
	local pingMs = Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
	local pingThreshold = 0.5 - (pingMs / 1000)
	local timeToArrival = distToBall / speed - (pingMs / 1000)
	local parryRadius = 25 - math.min(distToBall / 1000, 25) + speedBonus
	local clampedDot = math.clamp(dotProduct, - 1, 1)
	local targetRad = math.rad(math.asin(clampedDot))
	ballProps.lerp_radians = linearPredict(ballProps.lerp_radians, targetRad, 0.8)
	if speed > 100 and timeToArrival > pingMs / 10 then
		parryRadius = math.max(parryRadius - 15, 15)
	end
	if distToBall < parryRadius then
		return false
	end
	if curveFactor < pingThreshold then
		return true
	end
	if ballProps.lerp_radians < 0.018 then
		ballProps.last_warping = tick()
	end
	if (tick() - ballProps.last_warping) < (timeToArrival / 1.5) then
		return true
	end
	if (tick() - ballProps.curving) < (timeToArrival / 1.5) then
		return true
	end
	return dotProduct < pingThreshold
end

local spamRadius = 0
function Util.perfom_spam(params)
	local ballProps = Util.ball.properties
	local ball = Util.get_ball()
	local target = Util.closest_player()
	if not ball then
		return false
	end
	if not target or not target.PrimaryPart then
		return false
	end
	local velocity = ball.AssemblyLinearVelocity
	local speed = velocity.Magnitude
	local toPlayer = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Unit
	local dot = toPlayer:Dot(velocity.Unit)
	local targetPos = target.PrimaryPart.Position
	local distToTarget = LocalPlayer:DistanceFromCharacter(targetPos)
	local baseRadius = params.Ping + math.min(speed / 6, 6)
	if params.Entity_Properties.distance > baseRadius then
		return spamRadius
	end
	if params.Ball_Properties.Distance > baseRadius then
		return spamRadius
	end
	if distToTarget > baseRadius then
		return spamRadius
	end
	local dotBonus = 5 - math.min(speed / 5, 5)
	local dotPenalty = math.clamp(dot, - 1, 0) * dotBonus
	spamRadius = baseRadius - dotPenalty
	return spamRadius
end

MainTab:AddToggle("Toggle", {
	Title = "auto parry",
	Default = _G.config.auto_parry or false,
	Callback = function(enabled)
		_G.config.auto_parry = enabled
		if enabled then
			activeConnections.auto_parry = RunService.Heartbeat:Connect(function()
				local ballProps = Util.ball.properties
				local primaryBall = Util.get_ball()
				local allBalls = Util.get_balls()
				for _, ball in allBalls do
					if not ball then
						return
					end
					local zoomies = ball:FindFirstChild("zoomies")
					if not zoomies then
						return
					end
					ball:GetAttributeChangedSignal("target"):Once(function()
						isOnParryCooldown = false
					end)
					if isOnParryCooldown then
						return
					end
					local ballTarget = ball:GetAttribute("target")
					local primaryTarget = primaryBall:GetAttribute("target")
					local velocity = zoomies.VectorVelocity
					local distToBall = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Magnitude
					local pingValue = Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 10
					local pingClamp = math.clamp(pingValue / 10, 5, 17)
					local speed = velocity.Magnitude
					local speedAdjust = math.min(math.max(speed - 9.5, 0), 650)
					local timingBase = 2.4 + speedAdjust * 0.002
					local timingScale = accuracyMultiplier
					local timingFinal = timingBase * accuracyMultiplier
					local parryDistance = pingClamp + math.max(speed / timingFinal, 9.5)
					local isCurved = Util.ball_curved()
					if ball:FindFirstChild("AeroDynamicSlashVFX") then
						Debris:AddItem(ball.AeroDynamicSlashVFX, 0)
						ballProps.aerodynamic_time = tick()
					end
					if workspace.Runtime:FindFirstChild("Tornado") then
						local tornadoTime = workspace.Runtime.Tornado:GetAttribute("TornadoTime") or 1
						if (tick() - ballProps.aerodynamic_time) < tornadoTime + 0.314159 then
							return
						end
					end
					if primaryTarget == LocalPlayer.Name and isCurved then
						return
					end
					if ballTarget == LocalPlayer.Name and distToBall <= parryDistance and timingScale * 0.8 then
						Util.parry(_G.config.curve_method)
						isOnParryCooldown = true
					end
					local cooldownStart = tick()
					repeat
						RunService.PreSimulation:Wait()
					until (tick() - cooldownStart) >= 1 or not isOnParryCooldown
					isOnParryCooldown = false
				end
			end)
		else
			if activeConnections.auto_parry then
				activeConnections.auto_parry:Disconnect()
				activeConnections.auto_parry = nil
			end
		end
	end
})

-- Avatar Changer
-- Usa BindableEvent para despachar a aplicacao de skin para
-- a thread principal do script (identity alta), contornando
-- os callbacks de UI do Fluent que rodam com identity baixa.
local avatarPlr = game.Players.LocalPlayer
local avatarDesc = nil
local avatarCon = nil
local avatarOriginalDesc = nil
local avatarEvent = Instance.new("BindableEvent")
local avatarPendingName = nil

-- Busca a skin original do player ao iniciar
pcall(function()
	avatarOriginalDesc = game.Players:GetHumanoidDescriptionFromUserId(avatarPlr.UserId)
end)

-- Esta funcao aplica a skin e SEMPRE roda na thread do BindableEvent
-- que foi conectado aqui no top-level, mantendo a identity alta
local avatarLoopConn = nil
local function avatarApplyToChar(char)
	if not char or not avatarDesc then
		return
	end
	local hum = char:WaitForChild("Humanoid", 3)
	if not hum then
		return
	end
	avatarPlr:ClearCharacterAppearance()
	hum:ApplyDescriptionClientServer(avatarDesc)
end
avatarEvent.Event:Connect(function(action, data)
	if action == "apply" then
        -- Busca userId e desc
		local ok = pcall(function()
			local userId
			local asNum = tonumber(data)
			if asNum then
				userId = asNum
			else
				userId = game.Players:GetUserIdFromNameAsync(data)
			end
			if not userId then
				return
			end
			local desc = game.Players:GetHumanoidDescriptionFromUserId(userId)
			if not desc then
				return
			end
			avatarDesc = desc

            -- Aplica imediatamente no personagem atual
			avatarApplyToChar(avatarPlr.Character)
		end)

        -- Cancela conexoes antigas
		if avatarCon then
			avatarCon:Disconnect();
			avatarCon = nil
		end
		if avatarLoopConn then
			avatarLoopConn:Disconnect();
			avatarLoopConn = nil
		end

        -- Aplica ao nascer (lobby, respawn, etc.)
		avatarCon = avatarPlr.CharacterAdded:Connect(function(char)
			avatarEvent:Fire("respawn", char)
		end)

        -- Loop persistente: reaplica a skin a cada 2 segundos
        -- Garante que a skin fique mesmo em lobby ou apos reloads
		local lastApply = 0
		avatarLoopConn = RunService.Heartbeat:Connect(function()
			if not _G.config.avatar_changer or not avatarDesc then
				return
			end
			if (tick() - lastApply) < 2 then
				return
			end
			lastApply = tick()
			avatarEvent:Fire("reapply")
		end)
	elseif action == "respawn" then
		local char = data
		if not char or not avatarDesc then
			return
		end
		local hum = char:WaitForChild("Humanoid", 3)
		if not hum then
			return
		end
		avatarPlr:ClearCharacterAppearance()
		hum:ApplyDescriptionClientServer(avatarDesc)
	elseif action == "reapply" then
        -- Reaplica silenciosamente na thread de alta identity
		local char = avatarPlr.Character
		if not char or not avatarDesc then
			return
		end
		local hum = char:FindFirstChildOfClass("Humanoid")
		if not hum then
			return
		end
        -- Verifica se a skin ja foi aplicada antes de reaplicar
		local ok = pcall(function()
			local current = hum:GetAppliedDescription()
			if current and current.Face ~= avatarDesc.Face then
				avatarPlr:ClearCharacterAppearance()
				hum:ApplyDescriptionClientServer(avatarDesc)
			end
		end)
	elseif action == "stop" then
		if avatarCon then
			avatarCon:Disconnect();
			avatarCon = nil
		end
		if avatarLoopConn then
			avatarLoopConn:Disconnect();
			avatarLoopConn = nil
		end
		avatarDesc = nil
		local char = avatarPlr.Character
		if char and avatarOriginalDesc then
			local hum = char:FindFirstChildOfClass("Humanoid")
			if hum then
				avatarPlr:ClearCharacterAppearance()
				hum:ApplyDescriptionClientServer(avatarOriginalDesc)
			end
		end
	end
end)

-- Visualiser
local visualizerEnabled = false
local visualizerPart = nil

local function isSpamModeActive()
    return _G.AutoSpamValue or _G.manual_spam or false
end

local function calculate_visualizer_radius(ball)
    if not ball then return 0 end
    local zoomies = ball:FindFirstChild("zoomies")
    if not zoomies then return 0 end
    
    local velocity = zoomies.VectorVelocity
    local speed = velocity.Magnitude
    
    -- Dynamically mirror auto-parry activation calculations
    local pingValue = Stats.Network.ServerStatsItem:GetValue()
    local pingClamp = math.clamp((pingValue / 10) / 10, 5, 17)
    local speedAdjust = math.min(math.max(speed - 9.5, 0), 650)
    
    local timingBase = 2.4 + speedAdjust * 0.002
    local timingScale = accuracyMultiplier
    local timingFinal = timingBase * timingScale
    local parryDistance = pingClamp + math.max(speed / timingFinal, 9.5)
    
    -- Fallback to constant execution radius during active target clash
    if isSpamModeActive() then
        return 25
    end
    
    return parryDistance
end

local function createVisualizerPart()
    if visualizerPart and visualizerPart.Parent then
        pcall(function() visualizerPart:Destroy() end)
    end
    
    visualizerPart = Instance.new("Part")
    visualizerPart.Name = "ParryVisualizerSphere"
    visualizerPart.Shape = Enum.PartType.Ball
    visualizerPart.Anchored = true
    visualizerPart.CanCollide = false
    visualizerPart.CanTouch = false
    visualizerPart.CanQuery = false
    visualizerPart.CastShadow = false
    visualizerPart.Material = Enum.Material.ForceField
    visualizerPart.Transparency = 0.8
    visualizerPart.Color = Color3.fromRGB(240, 240, 245)
    
    -- Attach to local viewport container to protect from cleanup routines
    visualizerPart.Parent = workspace.CurrentCamera
end

RunService.RenderStepped:Connect(function()
    if not visualizerEnabled then 
        if visualizerPart and visualizerPart.Size ~= Vector3.zero then
            visualizerPart.Size = Vector3.zero
        end
        return 
    end
    
    -- Instantiate replacement if reference is swept
    if not visualizerPart or not visualizerPart.Parent or visualizerPart.Parent ~= workspace.CurrentCamera then
        createVisualizerPart()
    end
    
    local char = LocalPlayer.Character
    local primaryPart = char and char.PrimaryPart
    local ball = Util.get_ball()
    
    if not (primaryPart and ball) then
        visualizerPart.Size = Vector3.zero
        return
    end
    
    local target = ball:GetAttribute("target")
    local isTargetingPlayer = (target == LocalPlayer.Name)
    local radius = calculate_visualizer_radius(ball)
    
    -- Correct for standard diameter scaling properties
    local diameter = radius * 2
    visualizerPart.Size = Vector3.new(diameter, diameter, diameter)
    visualizerPart.CFrame = primaryPart.CFrame
    
    -- Dynamically update sphere color based on real-time threat levels
    if isSpamModeActive() or isTargetingPlayer then
        visualizerPart.Color = Color3.fromRGB(255, 0, 0)
        visualizerPart.Transparency = 0.65
    else
        visualizerPart.Color = Color3.fromRGB(240, 240, 245)
        visualizerPart.Transparency = 0.85
    end
end)

-- Korblox and Headless
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local KORBLOX_ASSET_ID = 139607718
local HEADLESS_ASSET_ID = 15093053680
local korbloxRespawnConn
local korbloxHeadlessConnection
local function updateKorbloxHeadless()
	local character = LocalPlayer.Character
	if not character then
		return
	end
    
    -- Headless
	local head = character:FindFirstChild("Head")
	if head then
		if head.Transparency ~= 1 then
			head.Transparency = 1
		end
		local face = head:FindFirstChild("face")
		if face and face:IsA("Decal") then
			if face.Transparency ~= 1 then
				face.Transparency = 1
			end
		end
	end
    
    -- Korblox Right Leg (R15)
	local r15Parts = {
		"RightUpperLeg",
		"RightLowerLeg",
		"RightFoot"
	}
	for _, partName in ipairs(r15Parts) do
		local part = character:FindFirstChild(partName)
		if part and part:IsA("BasePart") then
			if part.Transparency ~= 1 then
				part.Transparency = 1
			end
		end
	end
    
    -- Korblox Right Leg (R6)
	local r6Part = character:FindFirstChild("Right Leg")
	if r6Part and r6Part:IsA("BasePart") then
		if r6Part.Transparency ~= 1 then
			r6Part.Transparency = 1
		end
	end
end
local function restoreKorbloxHeadless()
	local character = LocalPlayer.Character
	if not character then
		return
	end
    
    -- Headless restore
	local head = character:FindFirstChild("Head")
	if head then
		head.Transparency = 0
		local face = head:FindFirstChild("face")
		if face and face:IsA("Decal") then
			face.Transparency = 0
		end
	end
    
    -- Korblox Right Leg (R15) restore
	local r15Parts = {
		"RightUpperLeg",
		"RightLowerLeg",
		"RightFoot"
	}
	for _, partName in ipairs(r15Parts) do
		local part = character:FindFirstChild(partName)
		if part and part:IsA("BasePart") then
			part.Transparency = 0
		end
	end
    
    -- Korblox Right Leg (R6) restore
	local r6Part = character:FindFirstChild("Right Leg")
	if r6Part and r6Part:IsA("BasePart") then
		r6Part.Transparency = 0
	end
end
local function applyKorbloxHeadless(enable)
	if enable then
		if not korbloxHeadlessConnection then
			korbloxHeadlessConnection = RunService.RenderStepped:Connect(updateKorbloxHeadless)
		end
	else
		if korbloxHeadlessConnection then
			korbloxHeadlessConnection:Disconnect()
			korbloxHeadlessConnection = nil
		end
		restoreKorbloxHeadless()
	end
end
local function startKorbloxLoop()
	if korbloxRespawnConn then
		korbloxRespawnConn:Disconnect();
		korbloxRespawnConn = nil
	end
	korbloxRespawnConn = LocalPlayer.CharacterAdded:Connect(function(char)
		local humanoid = char:WaitForChild("Humanoid", 5)
		if not humanoid then
			return
		end
		task.wait(1)
		applyKorbloxHeadless(true)
	end)
end
local function stopKorbloxLoop()
	if korbloxRespawnConn then
		korbloxRespawnConn:Disconnect();
		korbloxRespawnConn = nil
	end
end
VisualTab:AddToggle("KorbloxHeadless", {
	Title = "korblox + headless",
	Default = _G.config.korblox_headless or false,
	Callback = function(enabled)
		_G.config.korblox_headless = enabled
		task.spawn(function()
			if enabled then
				applyKorbloxHeadless(true)
				startKorbloxLoop()
			else
				stopKorbloxLoop()
				applyKorbloxHeadless(false)
			end
		end)
	end
})

		end
